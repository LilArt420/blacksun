// ====================================================================
// Black Sun -- a party instance on Mt. Mjolnir
// By: Lil Art
//
// James Wolf was bitten by a werewolf in the mountains. His older brother John
// (Prontera) fears what he will become when the Black Sun rises; their cousin
// Nightwing, who carries the Druids' gift and can take a wereraptor's shape,
// is already on the mountain.
//
// - Solo or party, and a real rAthena instance: every party (or solo player)
//   gets its own copy of the two maps, so any number of groups can run it at
//   once. The party leader (or the solo player) opens it at John Wolf; the
//   members join through him too, and can come back after dying.
// Every stage spawns all of its beasts at once when it starts, and each holds
// more than the one before. Every beast takes 10% of the damage it's dealt.
// - 1@mjo1, the Crag:
//     Horde 1, the Pack: 210 each of Moonfang Wolf, Black Sun Shade and
//       Frostbite Hound; 150 of each to kill
//     the three Alpha Werewolves and their packs (Nightwing turns into a
//       wereraptor)
//     Horde 2, the Black Sun rising: 420 each of Werewolf, Feral Wereraptor
//       and Black Sun Shade; 300 of each to kill
//     James, who throws everyone down the far side of the ridge
// - 1@mjo2, the Climb: 700 each of Werewolf, Frostbite Hound and Moonfang
//   Wolf scattered over the whole mountainside, frozen and staring until the
//   climbers have landed (320, 79); 500 of each open the summit
// - Niflheim: James surrenders to the Black Sun and becomes the MVP,
//   James, the Black Sun (2,000,000,000 HP, takes 1% damage). While anyone is
//   near him, the Black Sun calls 35 beasts every 15 seconds (his werewolf face
//   flashes up), and 3 Werewolves halfway between. When he falls, the chest of
//   spoils appears, and James comes back to himself once it has been opened.
// - Whenever there's someone to find, Nightwing says where, with a navigation
//   link (BS_Point).
// - 1.2.0, the hunt: Moon-Crowned beasts (2 of each kind in Horde 1, Horde 2
//   and the Climb, gold-crowned, 4x the HP; each counts for 10 and Nightwing
//   blesses the party), the Black Sun's Gaze (every 25s during the hordes, the
//   Climb and the finale the ground under each fighter is marked and erupts
//   3s later: step out), the Stranded Hunters (the old hunters of Mjolnir, 5
//   to rescue across the hordes and the Climb with Nightwing's links; each one
//   blesses the party and adds to the spoils), the Hunter's Bells (at 66% and
//   33% the eclipse shields James until the 2 true bells of the 4 around the
//   field are rung, 2 seconds standing still each; the other 2 are cursed and
//   turn into blue werewolves), and a grade for every night (time and falls:
//   S, A, B, C), with the fastest one on John's Hall of Hunters.
// - His reward is the Black Sun's, in a chest where he falls: base and job
//   EXP (10% of a level each), a Grey Wolf Cub in Mouth, a Wolf Orb (Above
//   All) Lv 3, 5 King Wolf Scrolls, 500 Premium Black Coins and a Premium
//   Black Ticket for everyone in the run; an 8 hour cooldown (per character).
//   90 minutes. Base level 200.
// ====================================================================

// The instance id of the attached character's run: their party's, or their own
function	script	BS_RunID	{
	if (getcharid(1)) return instance_id(IM_PARTY);
	return instance_id(IM_CHAR);
}

// 1 when the attached character may drive the story in instance <id>: the
// one who opened it, or their party's current leader. Party members are told
// so, anyone else is ignored.
function	script	BS_IsOwner	{
	.@iid = getarg(1);
	if (getcharid(0) == getinstancevar('bs_owner, .@iid)) return 1;
	.@pid = getinstancevar('bs_party, .@iid);
	if (.@pid && getcharid(1) == .@pid && getpartyleader(.@pid, 2) == getcharid(0)) return 1;
	if (.@pid && getcharid(1) == .@pid) {
		mes "[" + getarg(0) + "]";
		mes "Only your party leader can speak for the party up here.";
		close2;
	}
	return 0;
}

// Portrait pop-ups (cutin). The mod's own (data/texture/ui/illust/):
// 1 Nightwing (Royal Nightmare's), 4 John, and James: 5 himself, 6 the wolf
// breaking through, 7 the wolf winning, 3 the Black Sun's werewolf; 8 the
// Alpha Werewolves (the
// client's Druid werewolf, mirrored to face Nightwing). The client's own Druid
// art: 2 Nightwing as a wereraptor. 0 hides it.
function	script	BS_Portrait	{
	.@id = getarg(0);
	if (.@id == 1)      cutin "bs_nightwing.bmp", 2;
	else if (.@id == 2) cutin "druid_vedr.png", 2;
	else if (.@id == 3) cutin "bs_james_wolf.bmp", 2;	// James, the Black Sun
	else if (.@id == 4) cutin "bs_john.bmp", 0;
	else if (.@id == 5) cutin "bs_james.bmp", 2;		// James, himself
	else if (.@id == 6) cutin "bs_james_feral.bmp", 2;	// James, the wolf breaking through
	else if (.@id == 7) cutin "bs_james_feral2.bmp", 2;	// James, the wolf winning
	else if (.@id == 8) cutin "bs_alpha_varg.png", 0;	// the Alpha Werewolves, facing Nightwing
	else cutin "", 255;
	return;
}

// Nightwing's wereraptor shape. The Druid's own Wereraptor status carries a
// player-only script, so the server won't put it on an NPC: db/status.yml
// gives SC_FLOWERSMOKE (used by nothing else here) the wereraptor's icon, and
// the client draws whoever has it as the Druid's wereraptor form.
// The beasts are monsters with sprites of their own (data/sprite/monster/,
// System/jobname.lub), made from the Druid's forms, so they can be clicked
// and targeted like any monster. Every monster of the run spawns large.

// A navigation link to a spot on one of the run's maps, which the client knows
// by their own names (1@mjo1, 1@mjo2, nif_fild03). Clicking it opens the
// navigation; the coordinates are never shown.
//   callfunc("BS_Navi", "<text>", "<map>", <x>, <y>)
function	script	BS_Navi	{
	return "<NAVI>" + getarg(0) + "<INFO>" + getarg(1) + "," + getarg(2) + "," + getarg(3) + ",0,000,0</INFO></NAVI>";
}

// mm:ss for <seconds>
function	script	BS_Clock	{
	.@t = getarg(0);
	return (.@t / 60) + ":" + ((.@t % 60) < 10 ? "0" : "") + (.@t % 60);
}

// The Stranded Hunters: the old hunters of Mjolnir who warned of the Black Sun

	setarray .@n$[1], "Old Bram", "Ylva the Huntress", "Torvald the Archer", "Young Eir", "Sigrun the Ranger";
	return .@n$[getarg(0)];
}

// The Black Sun's spoils for the attached character, and their 8 hour
// cooldown: base and job EXP (10% of their current level each) and five
// items, put in their bag -- or, with getarg(0) = 1, sent by mail (the run is
// closing on someone who never opened the chest).
//   callfunc "BS_GiveRewards", <mail 0|1>, "<grade>", <hunters saved>, <night's seconds>;
// The grade adds coins (S 250 and a Ticket, A 150, B 50); every hunter saved
// 100 coins, and all five a Ticket.
function	script	BS_GiveRewards	{
	.@grade$ = getarg(1, "");
	.@hunters = getarg(2, 0);
	.@coins = 500 + .@hunters * 100;
	.@tickets = 1;
	if (.@hunters >= 5) .@tickets++;
	if (.@grade$ == "S") {
		.@coins += 250;
		.@tickets++;
	} else if (.@grade$ == "A") .@coins += 150;
	else if (.@grade$ == "B") .@coins += 50;
	if (getarg(3, 0) > 0 && (!BS_Best || getarg(3) < BS_Best)) BS_Best = getarg(3);
	setarray .@id[0], 420388, 310581, 22510, 1001258, 1002640;	// Grey Wolf Cub in Mouth, Wolf Orb (Above All) Lv 3, King Wolf Scroll, Premium Black Coin, Premium Black Ticket
	setarray .@n[0],       1,      1,     5,  .@coins, .@tickets;
	.@base = getbaseexp_ratio(10);
	.@job = getjobexp_ratio(10);
	if (.@base || .@job) getexp .@base, .@job;
	if (getarg(0, 0)) {
		mail getcharid(0), "John Wolf", "The Black Sun's spoils", "You left the mountain before you opened the chest where James fell. Here is your share. Thank you, for my brother. -- John", 0, .@id, .@n;
		dispbottom "[Black Sun] The Black Sun's spoils were sent to your mailbox.";
	} else {
		for (.@i = 0; .@i < 5; .@i++) getitem .@id[.@i], .@n[.@i];
		dispbottom "[Black Sun] You took the Black Sun's spoils: a Grey Wolf Cub in Mouth, a Wolf Orb (Above All) Lv 3, 5 King Wolf Scrolls, " + .@coins + " Premium Black Coins and " + .@tickets + " Premium Black Ticket(s).";
	}
	BS_Cooldown = gettimetick(2) + 28800;
	return;
}

		if (unitexists(.@a[.@i])) unittalk .@a[.@i], "Join the pack!";

L_Kill:
	if ('bs_stage != 1 && 'bs_stage != 5 && 'bs_stage != 7) return;
	.@t = getarg(0);
	callsub L_Cfg, 'bs_cfg;
	'bs_k[.@t] += getarg(1, 1);	// a Moon-Crowned beast counts for 10
	if ('bs_k[.@t] > .goal[.@t]) 'bs_k[.@t] = .goal[.@t];
	if (playerattached())
		dispbottom "[Black Sun] " + .nm$[0] + " " + 'bs_k[0] + "/" + .goal[0] + " | " + .nm$[1] + " " + 'bs_k[1] + "/" + .goal[1] + " | " + .nm$[2] + " " + 'bs_k[2] + "/" + .goal[2];
	for (.@i = 0; .@i < 3; .@i++)
		if ('bs_k[.@i] < .goal[.@i]) return;
	if ('bs_stage == 7) {
		callsub L_ClimbDone;
		return;
	}
	killmonster instance_mapname("1@mjo1"), "All";
	if ('bs_stage == 1) {
		'bs_stage = 2;
		callsub L_NightwingToLeader;
		instance_announce 0, "[Nightwing]: The Pack is breaking! Over here, quickly!", bc_map, "0x99CCFF";
	} else {
		'bs_stage = 6;
		.@m$ = instance_mapname("1@mjo1");
		.@jx = 199;
		.@jy = 230;
		getmapxy(.@pm$, .@px, .@py, BL_PC, 'bs_owner$);

		unitwarp getnpcid(0, .@jn$), .@m$, .@jx, .@jy;
		hideoffnpc .@jn$;
		callsub L_NightwingToLeader;
		instance_announce 0, "[Nightwing]: There! James! Talk to him, before the Black Sun does!", bc_map, "0x99CCFF";
	}
	return;

	// turned to stare at it
	'bs_wake = gettimetick(2) + 30;
	'bs_gaunN = 0;
	freeloop 1;	// 2,100 beasts to place and freeze in one go
	for (.@t = 0; .@t < 4; .@t++) {
		.@id = (.@t < 3 ? .id[.@t] : 26706);
		.@n = (.@t < 3 ? .spawn[.@t] : 8);

			for (.@try = 0; .@try < 50; .@try++) {
				getfreecell(.@m$, .@x, .@y);
				if (distance(.@x, .@y, 320, 79) >= 20) break;

			callsub L_Freeze, $@mobid[0], 320, 79;

	}
	// the Moon-Crowned: 2 of each kind, frozen with the rest
	for (.@t = 0; .@t < 3; .@t++) {
		for (.@c = 0; .@c < 2; .@c++) {
			for (.@try = 0; .@try < 50; .@try++) {
				getfreecell(.@m$, .@x, .@y);
				if (distance(.@x, .@y, 320, 79) >= 20) break;
			}
			callsub L_MoonAt, .@m$, .@x, .@y, .@t;
			callsub L_Freeze, $@mobid[0], 320, 79;
		}

	freeloop 0;
	end;

// --- 1-second loop: the HUD, the transforms, the safe landing, the finale ---
OnTimer1000:
	.@rem = 'bs_end - gettimetick(2);
	if ('bs_end && .@rem <= 0) {
		stopnpctimer;
		end;	// the instance's own time limit closes it
	}
	'bs_tick++;

	// a new sky for everyone in the run, at every stage (set by the kill scripts,
	// sent from here: attaching players from inside a kill script loses that script)
	if ('bs_skySent != 'bs_sky * 100 + 'bs_stage) {
		'bs_skySent = 'bs_sky * 100 + 'bs_stage;
		callsub L_Sky;
	}

	// the Climb wakes on its own if nobody closed the introduction in 30s
	if ('bs_wake && gettimetick(2) >= 'bs_wake) callsub L_Wake;

	// the landing on the crag stays clear while the hordes are out
	if ('bs_stage == 1 || 'bs_stage == 5) callsub L_SafeZone;

	// Nightwing's blessing on the party (a Moon-Crowned fell, or a hunter was saved)
	if ('bs_bless) {
		'bs_bless = 0;
		setarray .@maps$[0], "1@mjo1", "1@mjo2", "nif_fild03";
		for (.@mp = 0; .@mp < 3; .@mp++) {
			.@n = getmapunits(BL_PC, instance_mapname(.@maps$[.@mp]), .@pc);
			for (.@i = 0; .@i < .@n; .@i++) {
				if (!attachrid(.@pc[.@i])) continue;
				if (Hp > 0) {
					sc_start SC_BLESSING, 120000, 10;
					sc_start SC_INCREASEAGI, 120000, 10;
					specialeffect2 EF_BLESSING;
				}
				detachrid;
			}
		}
	}

	// the Black Sun's Gaze: marks every 25s during the hordes, the Climb (once
	// it has woken) and the finale, and erupts 3s after
	.@gzMap$ = "";
	if ('bs_stage == 1 || 'bs_stage == 5) { .@gzMap$ = "1@mjo1"; .@gzL$ = "a"; }
	else if ('bs_stage == 7 && !'bs_wake) { .@gzMap$ = "1@mjo2"; .@gzL$ = "b"; }
	else if ('bs_stage == 10) { .@gzMap$ = "nif_fild03"; .@gzL$ = "c"; }
	if ('bs_gzHide && gettimetick(2) >= 'bs_gzHide) {
		'bs_gzHide = 0;
		setarray .@lt$[0], "a", "b", "c";
		for (.@l = 0; .@l < 3; .@l++)
			for (.@h = 0; .@h < 8; .@h++) hideonnpc instance_npcname("Black Sun's Gaze#BS" + .@lt$[.@l] + .@h);
	}
	if ('bs_gzBoom && gettimetick(2) >= 'bs_gzBoom) callsub L_GazeBoom;
	else if (.@gzMap$ != "" && gettimetick(2) >= 'bs_gzAt) callsub L_GazeMark, .@gzMap$, .@gzL$;
	else if (.@gzMap$ == "" && gettimetick(2) >= 'bs_gzAt) 'bs_gzAt = gettimetick(2) + 10;	// the first mark comes 10s into the next fight

	// the Stranded Hunters: Horde 1 (30%, 70%), Horde 2 (30%, 70%), the Climb
	// (50%); one left alone 3 minutes is lost
	if ('bs_stage == 1 || 'bs_stage == 5 || ('bs_stage == 7 && !'bs_wake)) {
		callsub L_Cfg, 'bs_cfg;
		.@pct = ('bs_k[0] + 'bs_k[1] + 'bs_k[2]) * 100 / (.goal[0] + .goal[1] + .goal[2]);
		if ('bs_stage == 1) {
			if (.@pct >= 30 && !('bs_hunterSeen & (1 << 1))) callsub L_HunterAppear, 1;
			if (.@pct >= 70 && !('bs_hunterSeen & (1 << 2))) callsub L_HunterAppear, 2;
		} else if ('bs_stage == 5) {
			if (.@pct >= 30 && !('bs_hunterSeen & (1 << 3))) callsub L_HunterAppear, 3;
			if (.@pct >= 70 && !('bs_hunterSeen & (1 << 4))) callsub L_HunterAppear, 4;
		} else {
			if (.@pct >= 50 && !('bs_hunterSeen & (1 << 5))) callsub L_HunterAppear, 5;
		}
	}
	for (.@h = 1; .@h <= 5; .@h++)
		if (('bs_hunterOut & (1 << .@h)) && gettimetick(2) >= 'bs_hunterT[.@h] + 180) {
			'bs_hunterOut = 'bs_hunterOut & ~(1 << .@h);
			hideonnpc instance_npcname(callfunc("BS_HunterName", .@h) + "#BSH" + .@h);
			instance_announce 0, "[Nightwing]: Too late... the beasts found " + callfunc("BS_HunterName", .@h) + ".", bc_map, "0xFF4444";
		}
	// Nightwing points everyone on the hunter's map to them
	if ('bs_hunterPoint) {
		'bs_hunterPoint = 0;
		.@hm$ = ('bs_hunterNow == 5 ? "1@mjo2" : "1@mjo1");
		.@n = getmapunits(BL_PC, instance_mapname(.@hm$), .@pc);
		for (.@i = 0; .@i < .@n; .@i++) {
			if (!attachrid(.@pc[.@i])) continue;
			addtimer 500, instance_npcname("BS_Point#" + ('bs_hunterNow == 5 ? 2 : 1)) + "::OnHunter";
			detachrid;
		}
	}

	// the Hunter's Bells: at 66% and 33% of his HP the eclipse shields James
	if ('bs_stage == 10 && 'bs_mvp && !'bs_shield && 'bs_shieldPhase < 2 && unitexists('bs_mvp)) {
		getunitdata 'bs_mvp, .@jd;
		if (.@jd[UMOB_MAXHP] >= 100) {
			.@pct = .@jd[UMOB_HP] / (.@jd[UMOB_MAXHP] / 100);	// 2 billion HP: never multiply it
			if (('bs_shieldPhase == 0 && .@pct <= 66) || ('bs_shieldPhase == 1 && .@pct <= 33)) callsub L_Shield;
		}
	}

	// the finale: every 15 seconds the Black Sun calls 35 beasts around one of
	// the fighters near James, picked at random (his werewolf face flashes up
	// for everyone on the field), and halfway between, 3 Werewolves
	if ('bs_cutOff && gettimetick(2) >= 'bs_cutOff) {
		'bs_cutOff = 0;
		callsub L_MapCutin, "nif_fild03", "", 255;
	}
	if ('bs_stage == 10 && 'bs_mvp) {
		.@m$ = instance_mapname("nif_fild03");
		.@wave = 0;
		if (gettimetick(2) >= 'bs_addTime) .@wave = 35;
		else if ('bs_wolfTime && gettimetick(2) >= 'bs_wolfTime) .@wave = 3;
		if (.@wave && unitexists('bs_mvp) && getmapunits(BL_MOB, .@m$) < 175) {
			getunitdata 'bs_mvp, .@md;
			.@n = getmapunits(BL_PC, .@m$, .@pc);
			.@k = 0;
			for (.@i = 0; .@i < .@n; .@i++) {
				getmapxy(.@pm$, .@px, .@py, BL_PC, rid2name(.@pc[.@i]));
				if (.@pm$ == .@m$ && distance(.@px, .@py, .@md[UMOB_X], .@md[UMOB_Y]) <= 14) {
					.@nx[.@k] = .@px;
					.@ny[.@k] = .@py;
					.@k++;
				}
			}
			if (.@k) {
				// rand(1) is an error that ends this loop, and rAthena's ?: works out
				// both sides, so this has to be an if
				.@r = 0;
				if (.@k > 1) .@r = rand(.@k);
				.@px = .@nx[.@r];
				.@py = .@ny[.@r];
				if (.@wave == 35) {
					callsub L_SpawnRing, .@m$, .@px, .@py, 7, 1022, 5, "";
					callsub L_SpawnRing, .@m$, .@px, .@py, 7, 26705, 7, "";
					callsub L_SpawnRing, .@m$, .@px, .@py, 5, 26702, 8, "";
					callsub L_SpawnRing, .@m$, .@px, .@py, 5, 26703, 8, "";
					callsub L_SpawnRing, .@m$, .@px, .@py, 9, 26704, 7, "";
					instance_announce 0, "[Black Sun]: The Black Sun calls its beasts to James's side!", bc_map, "0xFF6600";
					'bs_addTime = gettimetick(2) + 15;
					'bs_wolfTime = gettimetick(2) + 7;
					'bs_cutOff = gettimetick(2) + 3;
					callsub L_MapCutin, "nif_fild03", "bs_james_wolf.bmp", 2;
				} else {
					callsub L_SpawnRing, .@m$, .@px, .@py, 4, 1022, 3, "";
					'bs_wolfTime = 0;
				}
			} else if (.@wave == 35) {
				'bs_addTime = gettimetick(2) + 3;	// nobody near: look again soon
			}
		}
	}

	// James comes back to himself once everyone on the field has taken their
	// share from the chest (or a minute after the leader took theirs); the
	// chest asks too, the moment someone opens it
	if ('bs_stage == 11 && !'bs_jamesBack) callsub L_JamesBack;

	// Nightwing points the way whenever there's someone to find
	if ('bs_pointSent != 'bs_stage * 10 + 'bs_jamesBack) {
		'bs_pointSent = 'bs_stage * 10 + 'bs_jamesBack;
		if ('bs_stage == 2 || 'bs_stage == 4 || 'bs_stage == 6 || 'bs_stage == 8 || 'bs_stage == 9 || 'bs_stage == 11)
			callsub L_Point;
	}

	// the HUD
	.@time$ = (.@rem / 60) + ":" + ((.@rem % 60) < 10 ? "0" : "") + (.@rem % 60);
	switch ('bs_stage) {
	case 0:  .@info$ = "Talk to Nightwing"; break;
	case 1:
	case 5:
	case 7:
		callsub L_Cfg, 'bs_cfg;
		.@info$ = ('bs_stage == 7 ? "The Climb" : "Horde " + 'bs_cfg) + " | " + .ab$[0] + " " + 'bs_k[0] + "/" + .goal[0] + " " + .ab$[1] + " " + 'bs_k[1] + "/" + .goal[1] + " " + .ab$[2] + " " + 'bs_k[2] + "/" + .goal[2];
		break;
	case 2:
	case 4:  .@info$ = "Talk to Nightwing"; break;
	case 3:  .@info$ = "Alpha Werewolves " + 'bs_alphaK + "/3"; break;
	case 6:  .@info$ = "Talk to James"; break;
	case 8:  .@info$ = "Talk to Nightwing at the summit"; break;
	case 9:  .@info$ = "Talk to James"; break;
	case 10: .@info$ = "James, the Black Sun"; break;
	case 11: .@info$ = ('bs_jamesBack ? "Talk to James" : "Open the chest where James fell"); break;
	default: .@info$ = "Talk to James";
	}
	if ('bs_stage == 10 && 'bs_shield) .@info$ = "Ring the true Hunter's Bells " + 'bs_bells + "/2";
	if ('bs_stage >= 1) .@info$ += " | Hunters " + 'bs_hunters + "/5";
	setarray .@maps$[0], "1@mjo1", "1@mjo2", "nif_fild03";
	for (.@mp = 0; .@mp < 3; .@mp++) {
		.@n = getmapunits(BL_PC, instance_mapname(.@maps$[.@mp]), .@pc);
		for (.@i = 0; .@i < .@n; .@i++) {
			if (!attachrid(.@pc[.@i])) continue;
			showscript "[Black Sun] " + .@time$ + " | " + .@info$;
			if (.@rem == 1800 || .@rem == 600 || .@rem == 300 || .@rem == 60)
				dispbottom "[Black Sun] " + (.@rem / 60) + " minute(s) until the Black Sun is gone and the mountain closes!";
			detachrid;

// Freezes <gid> (mode 0: it can neither move nor attack; its mode is kept
// for L_Wake) and turns it towards <x>,<y>.
// Directions: 0 N, 1 NW, 2 W, 3 SW, 4 S, 5 SE, 6 E, 7 NE.
L_Freeze:
	.@g = getarg(0);
	// getunitdata is a command (it returns nothing): ask unitexists first
	if (!.@g || !unitexists(.@g)) return;
	getunitdata .@g, .@d;
	if ('bs_wake == 0) { 'bs_gaunN = 0; 'bs_wake = gettimetick(2) + 30; }
	'bs_gaunGID['bs_gaunN] = .@g;
	'bs_gaunMode['bs_gaunN] = .@d[UMOB_MODE];
	'bs_gaunN++;
	.@dx = getarg(1) - .@d[UMOB_X];
	.@dy = getarg(2) - .@d[UMOB_Y];
	.@ax = .@dx < 0 ? -.@dx : .@dx;
	.@ay = .@dy < 0 ? -.@dy : .@dy;
	if (.@ax * 2 < .@ay) .@dir = .@dy > 0 ? 0 : 4;
	else if (.@ay * 2 < .@ax) .@dir = .@dx > 0 ? 6 : 2;
	else if (.@dx > 0) .@dir = .@dy > 0 ? 7 : 5;
	else .@dir = .@dy > 0 ? 1 : 3;
	setunitdata .@g, UMOB_MODE, 0;
	setunitdata .@g, UMOB_LOOKDIR, .@dir;
	return;

// Every frozen beast gets its own mode back
L_Wake:
	if (!'bs_wake) return;
	'bs_wake = 0;
	freeloop 1;
	for (.@i = 0; .@i < 'bs_gaunN; .@i++)
		if (unitexists('bs_gaunGID[.@i]))
			setunitdata 'bs_gaunGID[.@i], UMOB_MODE, 'bs_gaunMode[.@i];
	'bs_gaunN = 0;
	deletearray 'bs_gaunGID;
	deletearray 'bs_gaunMode;
	freeloop 0;
	return;

// The Hunter's Bells around the field: when the eclipse shields James (66%
// and 33% of his HP), clicking one and standing still beside it for 2 seconds
// rings it. 2 of the 4 are true ('bs_bellTrue, new every time): the 2nd true
// one breaks the shield (BS_Engine::OnBellRung). The other 2 are cursed and
// turn into a blue werewolf (the Alpha Werewolf, the Druid's blue form).
// The bar is the bell's own (progressbar_npc), timed by the server: the
// player's own bar ends a hair early on clients before app 1.7.0, which the
// server counts as cancelled, and a hit in the fight cancels it.
nif_fild03,193,196,4	script	Hunter's Bell#BSB0	2917,{
	.@b = atoi(charat(strnpcinfo(2), 3));	// BSB<n>
	if ('bs_stage != 10 || !'bs_shield || ('bs_bellMask & (1 << .@b))) end;
	if (gettimetick(2) < 'bs_ringT[.@b]) end;	// someone is ringing it right now
	'bs_ringT[.@b] = gettimetick(2) + 3;
	progressbar_npc "0xFFFFFF", 2;	// holds the ringer still for 2 seconds
	'bs_ringT[.@b] = 0;
	if ('bs_stage != 10 || !'bs_shield || ('bs_bellMask & (1 << .@b)) || Hp < 1) end;
	'bs_bellMask = 'bs_bellMask | (1 << .@b);
	if ('bs_bellTrue & (1 << .@b)) {
		'bs_bells++;
		specialeffect EF_GLORIA, AREA, strnpcinfo(3);
		donpcevent instance_npcname("BS_Engine") + "::OnBellRung";
		end;
	}
	// a cursed bell: it cracks, and a blue werewolf stands where it was
	getmapxy(.@m$, .@x, .@y, BL_NPC);
	specialeffect EF_SUI_EXPLOSION, AREA, strnpcinfo(3);
	hideonnpc strnpcinfo(3);
	monster .@m$, .@x, .@y, "--ja--", 26701, 1, "", Size_Large;
	instance_announce 0, "[Black Sun]: A cursed bell! It cracks open, and a blue werewolf howls out of it!", bc_map, "0xFF0000";
	end;
OnInit:
	hideonnpc strnpcinfo(3);
	end;
OnInstanceInit:
	hideonnpc strnpcinfo(3);
	end;
}
